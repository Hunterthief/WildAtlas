    73→// Load model links from JSON file
    74→async function loadModelLinks() {
    75→    try {
    76→        const response = await fetch('data/model_links.json?t=' + Date.now());
    77→        if (response.ok) {
    78→            SKETCHFAB_MODELS = await response.json();
    79→            console.log(`✅ Loaded ${Object.keys(SKETCHFAB_MODELS).length} 3D model links`);
    80→        } else {
    81→            console.warn('⚠️ Could not load model_links.json - using images only');
    82→        }
    83→    } catch (error) {
    84→        console.error('❌ Error loading model links:', error);
    85→    }
    86→}

    88→// Find matching 3D model for animal
    89→function findMatchingModel(animalName) {
    90→    const nameLower = animalName.toLowerCase();
    91→    
    92→    // Direct match
    93→    if (SKETCHFAB_MODELS[nameLower]) {
    94→        return SKETCHFAB_MODELS[nameLower];
    95→    }
    96→    
    97→    // Partial match
    98→    for (const [key, url] of Object.entries(SKETCHFAB_MODELS)) {
    99→        if (nameLower.includes(key) || key.includes(nameLower)) {
   100→            return url;
   101→        }
   102→    }
   103→    
   104→    return null;
   105→}

   107→// Setup 3D model viewer (or show image if no model)
   108→async function setup3DModel(animal) {
   109→    const modelViewer = document.getElementById('animal-3d-model');
   110→    const imageElement = document.getElementById('animal-image');
   111→    const toggleButton = document.getElementById('view-toggle');
   112→    const toggleText = document.getElementById('view-toggle-text');
   113→    
   114→    if (!modelViewer || !imageElement) return;
   115→    
   116→    // Find matching 3D model
   117→    const modelUrl = findMatchingModel(animal.name);
   118→    
   119→    // No 3D model found - show Wikipedia image only
   120→    if (!modelUrl) {
   121→        console.log(`ℹ️ No 3D model for ${animal.name} - using Wikipedia image`);
   122→        modelViewer.style.display = 'none';
   123→        imageElement.style.display = 'block';
   124→        if (toggleButton) toggleButton.style.display = 'none';
   125→        return;
   126→    }
   127→    
   128→    // 3D model found - load it
   129→    try {
   130→        const oembedUrl = `https://sketchfab.com/oembed?url=${encodeURIComponent(modelUrl)}&maxwidth=800`;
   131→        const response = await fetch(oembedUrl);
   132→        
   133→        if (response.ok) {
   134→            const data = await response.json();
   135→            
   136→            if (data.glb_url || data.gltf_url) {
   137→                // Load 3D model
   138→                modelViewer.src = data.glb_url || data.gltf_url;
   139→                modelViewer.alt = `3D model of ${animal.name}`;
   140→                modelViewer.style.display = 'block';
   141→                
   142→                // Hide image initially
   143→                imageElement.style.display = 'none';
   144→                
   145→                // Show toggle button
   146→                if (toggleButton) {
   147→                    toggleButton.style.display = 'flex';
   148→                    toggleText.textContent = 'View Image';
   149→                }
   150→                
   151→                // Setup toggle between 3D and image
   152→                setupViewToggle(modelViewer, imageElement, toggleButton, toggleText);
   153→                console.log(`✅ 3D model loaded for ${animal.name}`);
   154→                return;
   155→            }
   156→        }
   157→    } catch (error) {
   158→        console.error(`❌ Error loading 3D model for ${animal.name}:`, error);
   159→    }
   160→    
   161→    // Fallback to image if model fails to load
   162→    console.log(`⚠️ 3D model failed for ${animal.name} - using image`);
   163→    modelViewer.style.display = 'none';
   164→    imageElement.style.display = 'block';
   165→    if (toggleButton) toggleButton.style.display = 'none';
   166→}

   211→function setupSectionToggles() {
   212→    const sectionHeaders = document.querySelectorAll('.section-header');
   213→    
   214→    sectionHeaders.forEach(header => {
   215→        header.addEventListener('click', () => {
   216→            const section = header.parentElement;
   217→            section.classList.toggle('collapsed');
   218→            
   219→            // Save state to localStorage
   220→            const type = section.dataset.type;
   221→            localStorage.setItem(`section-${type}`, section.classList.contains('collapsed') ? 'collapsed' : 'expanded');
   222→        });
   223→    });
   224→    
   225→    // Restore state from localStorage
   226→    const types = ['mammal', 'bird', 'reptile', 'fish', 'amphibian', 'insect'];
   227→    types.forEach(type => {
   228→        const state = localStorage.getItem(`section-${type}`);
   229→        if (state === 'collapsed') {
   230→            const section = document.querySelector(`.type-section[data-type="${type}"]`);
   231→            if (section) {
   232→                section.classList.add('collapsed');
   233→            }
   234→        }
   235→    });

   270→    // Restore preference

   273→        const grids = document.querySelectorAll('.animal-grid');
   274→        grids.forEach(grid => grid.classList.add('list-view'));

   280→                btn.classList.remove('active');

   319→// Organize animals by type into sections
   320→function organizeByType(animals) {
   321→    const typeMappings = {
   322→        'mammal': ['mammal', 'mammalia', 'feline', 'canine', 'bear', 'elephant', 'primate', 'whale', 'deer', 'bovine', 'equine', 'rabbit', 'rodent', 'bat', 'giraffe', 'cheetah'],
   323→        'bird': ['bird', 'aves', 'raptor', 'owl', 'penguin', 'chicken', 'duck', 'goose', 'swan', 'eagle'],
   324→        'reptile': ['reptile', 'reptilia', 'snake', 'lizard', 'turtle', 'crocodile'],
   325→        'fish': ['fish', 'shark', 'ray', 'salmon'],
   326→        'amphibian': ['amphibian', 'amphibia', 'frog', 'salamander'],
   327→        'insect': ['insect', 'insecta', 'butterfly', 'bee', 'ant', 'spider', 'crab']
   328→    };
   329→    
   330→    // Clear all grids
   331→    Object.keys(typeMappings).forEach(type => {
   332→        document.getElementById(`grid-${type}`).innerHTML = '';
   333→    });
   334→    
   335→    // Count animals per type
   336→    const counts = {
   337→        mammal: 0,
   338→        bird: 0,
   339→        reptile: 0,
   340→        fish: 0,
   341→        amphibian: 0,
   342→        insect: 0
   343→    };
   344→    
   345→    // Organize animals
   346→    animals.forEach(animal => {
   347→        const animalType = animal.animal_type?.toLowerCase() || '';
   348→        const classType = animal.classification?.class?.toLowerCase() || '';
   349→        
   350→        // Find matching type
   351→        for (const [type, validTypes] of Object.entries(typeMappings)) {
   352→            if (validTypes.some(t => animalType.includes(t) || classType.includes(t))) {
   353→                renderAnimalCard(animal, type);
   354→                counts[type]++;
   355→                break;
   356→            }
   357→        }
   358→    });
   359→    
   360→    // Update counts
   361→    for (const [type, count] of Object.entries(counts)) {
   362→        const countElement = document.getElementById(`count-${type}`);
   363→        if (countElement) {
   364→            countElement.textContent = count;
   365→        }
   366→    }
   367→}

   397→// Filter animals across all sections
   398→function filterAnimals(query) {
   399→    const typeMappings = {
   400→        'mammal': ['mammal', 'mammalia', 'feline', 'canine', 'bear', 'elephant', 'primate', 'whale', 'deer', 'bovine', 'equine', 'rabbit', 'rodent', 'bat', 'giraffe', 'cheetah'],
   401→        'bird': ['bird', 'aves', 'raptor', 'owl', 'penguin', 'chicken', 'duck', 'goose', 'swan', 'eagle'],
   402→        'reptile': ['reptile', 'reptilia', 'snake', 'lizard', 'turtle', 'crocodile'],
   403→        'fish': ['fish', 'shark', 'ray', 'salmon'],
   404→        'amphibian': ['amphibian', 'amphibia', 'frog', 'salamander'],
   405→        'insect': ['insect', 'insecta', 'butterfly', 'bee', 'ant', 'spider', 'crab']
   406→    };
   407→    
   408→    if (!query) {
   409→        // Show all
   410→        organizeByType(allAnimals);
   411→        return;
   412→    }
   413→    
   414→    // Clear all grids
   415→    Object.keys(typeMappings).forEach(type => {
   416→        document.getElementById(`grid-${type}`).innerHTML = '';
   417→    });
   418→    
   419→    // Filter and render
   420→    const filtered = allAnimals.filter(animal => {
   421→        const nameMatch = animal.name?.toLowerCase().includes(query);
   422→        const scientificMatch = animal.scientific_name?.toLowerCase().includes(query);
   423→        const typeMatch = animal.animal_type?.toLowerCase().includes(query);
   424→        const habitatMatch = animal.ecology?.habitat?.toLowerCase().includes(query);
   425→        const locationMatch = animal.ecology?.locations?.toLowerCase().includes(query);
   426→        return nameMatch || scientificMatch || typeMatch || habitatMatch || locationMatch;
   427→    });
   428→    
   429→    // Re-organize filtered results
   430→    filtered.forEach(animal => {
   431→        const animalType = animal.animal_type?.toLowerCase() || '';
   432→        const classType = animal.classification?.class?.toLowerCase() || '';
   433→        
   434→        for (const [type, validTypes] of Object.entries(typeMappings)) {
   435→            if (validTypes.some(t => animalType.includes(t) || classType.includes(t))) {
   436→                renderAnimalCard(animal, type);
   437→                break;
   438→            }
   439→        }
   440→    });
   441→    
   442→    // Update counts
   443→    const counts = { mammal: 0, bird: 0, reptile: 0, fish: 0, amphibian: 0, insect: 0 };
   444→    filtered.forEach(animal => {
   445→        const animalType = animal.animal_type?.toLowerCase() || '';
   446→        const classType = animal.classification?.class?.toLowerCase() || '';
   447→        
   448→        for (const [type, validTypes] of Object.entries(typeMappings)) {
   449→            if (validTypes.some(t => animalType.includes(t) || classType.includes(t))) {
   450→                counts[type]++;
   451→                break;
   452→            }
   453→        }
   454→    });
   455→    
   456→    for (const [type, count] of Object.entries(counts)) {
   457→        const countElement = document.getElementById(`count-${type}`);
   458→        if (countElement) {
   459→            countElement.textContent = count;
   460→        }
   461→    }
   462→}

   775→function addLocationDots(locationsString) {
   776→    const dotsContainer = document.getElementById('location-dots');
   777→    if (!dotsContainer || !locationsString) return;
   778→    
   779→    dotsContainer.innerHTML = '';
   780→    
   781→    const locations = locationsString.toLowerCase().split(',').map(l => l.trim());
   782→    const addedDots = new Set();
   783→    
   784→    locations.forEach(location => {
   785→        const coords = findLocationCoordinates(location);
   786→        if (coords) {
   787→            const dotKey = `${coords.x}-${coords.y}`;
   788→            if (!addedDots.has(dotKey)) {
   789→                addedDots.add(dotKey);
   790→                
   791→                const dot = document.createElement('div');
   792→                dot.className = 'location-dot';
   793→                dot.style.left = coords.x + '%';
   794→                dot.style.top = coords.y + '%';
   795→                dot.setAttribute('data-label', location);
   796→                dot.setAttribute('title', location);
   797→                
   798→                dotsContainer.appendChild(dot);
   799→            }
   800→        }
   801→    });
   802→}

   804→function findLocationCoordinates(location) {
   805→    const loc = location.toLowerCase().trim();
   806→    
   807→    // Direct match
   808→    if (locationCoordinates[loc]) {
   809→        return locationCoordinates[loc];
   810→    }
   811→    
   812→    // Partial match
   813→    for (const key in locationCoordinates) {
   814→        if (loc.includes(key) || key.includes(loc)) {
   815→            return locationCoordinates[key];
   816→        }
   817→    }
   818→    
   819→    // Fallback guesses
   820→    if (loc.includes('asia') || loc.includes('east')) return locationCoordinates['asia'];
   821→    if (loc.includes('america') || loc.includes('usa') || loc.includes('us') || loc.includes('united states')) {
   822→        return locationCoordinates['north america'];
   823→    }
   824→    if (loc.includes('europe')) return locationCoordinates['europe'];
   825→    if (loc.includes('africa')) return locationCoordinates['africa'];
   826→    if (loc.includes('australia') || loc.includes('oceania')) return locationCoordinates['australia'];
   827→    if (loc.includes('south') && !loc.includes('korea')) return locationCoordinates['south america'];
   828→    if (loc.includes('india')) return locationCoordinates['india'];
   829→    if (loc.includes('china')) return locationCoordinates['china'];
   830→    if (loc.includes('russia')) return locationCoordinates['russia'];
   831→    if (loc.includes('indonesia')) return locationCoordinates['indonesia'];
   832→    
   833→    return null;
   834→}

   839→function getTimePeriod(animal) {

   847→            end: animal.time_period.end || 'Present'
   848→        };

   850→    
   851→    // FALLBACK: Class-based estimation (if no time_period in JSON)
   852→    const animalType = animal.animal_type?.toLowerCase() || '';
   853→    const classType = animal.classification?.class?.toLowerCase() || '';
   854→    
   855→    let millionsYears = 0;
   856→    let text = '';
   857→    let width = '50%';

   861→        animalType.includes('wolf') || animalType.includes('tiger')) {
   862→        millionsYears = 200;
   863→        text = `Evolved ~${millionsYears} million years ago`;
   864→        width = '85%';
   865→    }
   866→    else if (classType.includes('aves') || animalType.includes('bird') || animalType.includes('eagle') ||
   867→        animalType.includes('penguin') || animalType.includes('raptor')) {
   868→        millionsYears = 150;
   869→        text = `Evolved ~${millionsYears} million years ago`;
   870→        width = '75%';
   871→    }
   872→    else if (classType.includes('reptil') || animalType.includes('snake') || animalType.includes('turtle') ||
   873→        animalType.includes('cobra')) {
   874→        millionsYears = 300;
   875→        text = `Evolved ~${millionsYears} million years ago`;
   876→        width = '90%';
   877→    }
   878→    else if (classType.includes('fish') || classType.includes('chondrichthyes') || classType.includes('actinopterygii') ||
   879→        animalType.includes('shark') || animalType.includes('salmon')) {
   880→        millionsYears = 500;
   881→        text = `Evolved ~${millionsYears} million years ago`;
   882→        width = '95%';
   883→    }
   884→    else if (classType.includes('amphib') || animalType.includes('frog')) {
   885→        millionsYears = 370;
   886→        text = `Evolved ~${millionsYears} million years ago`;
   887→        width = '92%';
   888→    }
   889→    else if (classType.includes('insect') || animalType.includes('butterfly') || animalType.includes('bee')) {
   890→        millionsYears = 400;
   891→        text = `Evolved ~${millionsYears} million years ago`;
   892→        width = '93%';
   893→    }
   894→    else {
   895→        millionsYears = 100;
   896→        text = `Evolved ~${millionsYears} million years ago`;
   897→        width = '60%';
   898→    }
   899→    
   900→    return {
   901→        text: text,
   902→        width: width,
   903→        start: `${millionsYears}M years ago`,
   904→        end: 'Present'

   911→function generateDescriptionText(animal, summary) {
   912→    const phys = animal.physical || {};
   913→    const eco = animal.ecology || {};

   939→        if (sentences.length > 0) {
   940→            parts.push(sentences[0].trim() + '.');
   941→        }
   942→    }
   943→    
   944→    return parts.join(' ');
   945→}

  1169→function getDietTypes(diet, animalType, summary) {
  1170→    if (!diet) return [{ class: 'unknown', icon: '❓' }];
  1171→    
  1172→    const dietLower = diet.toLowerCase();
  1173→    const summaryLower = (summary || '').toLowerCase();
  1174→    const typeLower = (animalType || '').toLowerCase();
  1175→    
  1176→    const types = [];
  1177→    
  1178→    // CARNIVORE / MEAT
  1179→    if (dietLower.includes('carnivore') ||
  1180→        dietLower.includes('meat') ||
  1181→        summaryLower.includes('predator') ||
  1182→        summaryLower.includes('preys on') ||
  1183→        summaryLower.includes('hunts') ||
  1184→        ['feline', 'canine', 'bear', 'shark', 'raptor', 'snake', 'crocodile'].includes(typeLower)) {
  1185→        types.push({ class: 'carnivore', icon: '🥩', title: 'Meat' });
  1186→    }
  1187→    
  1188→    // HERBIVORE / PLANTS
  1189→    if (dietLower.includes('herbivore') ||
  1190→        dietLower.includes('plant') ||
  1191→        summaryLower.includes('grazes') ||
  1192→        summaryLower.includes('foliage') ||
  1193→        summaryLower.includes('vegetation') ||
  1194→        ['elephant', 'bovine', 'deer', 'rabbit', 'turtle'].includes(typeLower)) {
  1195→        types.push({ class: 'herbivore', icon: '🌿', title: 'Plants' });
  1196→    }
  1197→    
  1198→    // PISCIVORE / FISH
  1199→    if (dietLower.includes('piscivore') ||
  1200→        dietLower.includes('fish') ||
  1201→        summaryLower.includes('fish') ||
  1202→        summaryLower.includes('salmon') ||
  1203→        summaryLower.includes('marine') ||
  1204→        ['shark', 'eagle', 'penguin', 'bear', 'seal', 'otter'].includes(typeLower)) {
  1205→        types.push({ class: 'piscivore', icon: '🐟', title: 'Fish' });
  1206→    }
  1207→    
  1208→    // INSECTIVORE / INSECTS
  1209→    if (dietLower.includes('insectivore') ||
  1210→        summaryLower.includes('insects') ||
  1211→        summaryLower.includes('bugs') ||
  1212→        summaryLower.includes('arthropods') ||
  1213→        ['frog', 'bat', 'spider', 'lizard', 'bird'].includes(typeLower)) {
  1214→        types.push({ class: 'insectivore', icon: '🐛', title: 'Insects' });
  1215→    }
  1216→    
  1217→    // OMNIVORE / MIXED
  1218→    if (dietLower.includes('omnivore') ||
  1219→        summaryLower.includes('varied diet') ||
  1220→        summaryLower.includes('both plants and animals') ||
  1221→        ['bear', 'pig', 'raccoon', 'crow'].includes(typeLower)) {
  1222→        types.push({ class: 'omnivore', icon: '🍽️', title: 'Omnivore' });
  1223→    }
  1224→    
  1225→    // NECTARIVORE / NECTAR
  1226→    if (summaryLower.includes('nectar') ||
  1227→        summaryLower.includes('pollinator') ||
  1228→        ['butterfly', 'bee', 'hummingbird'].includes(typeLower)) {
  1229→        types.push({ class: 'nectarivore', icon: '🌸', title: 'Nectar' });
  1230→    }
  1231→    
  1232→    // SCAVENGER
  1233→    if (summaryLower.includes('scavenger') ||
  1234→        summaryLower.includes('carrion') ||
  1235→        ['vulture', 'hyena'].includes(typeLower)) {
  1236→        types.push({ class: 'scavenger', icon: '🦴', title: 'Scavenger' });
  1237→    }
  1238→    
  1239→    // FALLBACK
  1240→    if (types.length === 0) {
  1241→        if (dietLower.includes('carnivore')) {
  1242→            types.push({ class: 'carnivore', icon: '🥩', title: 'Carnivore' });
  1243→        } else if (dietLower.includes('herbivore')) {
  1244→            types.push({ class: 'herbivore', icon: '🌿', title: 'Herbivore' });
  1245→        } else {
  1246→            types.push({ class: 'omnivore', icon: '🍽️', title: 'Omnivore' });
  1247→        }
  1248→    }
  1249→    
  1250→    // Remove duplicates
  1251→    const unique = [];
  1252→    const seen = new Set();
  1253→    for (const t of types) {
  1254→        if (!seen.has(t.class)) {
  1255→            seen.add(t.class);
  1256→            unique.push(t);
  1257→        }
  1258→    }
  1259→    
  1260→    return unique;
  1261→}
