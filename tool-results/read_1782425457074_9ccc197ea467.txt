    88→    88→// Find matching 3D model for animal
    89→    89→function findMatchingModel(animalName) {
    90→    90→    const nameLower = animalName.toLowerCase();
    91→    91→    
    92→    92→    // Direct match
    93→    93→    if (SKETCHFAB_MODELS[nameLower]) {
    94→    94→        return SKETCHFAB_MODELS[nameLower];
    95→    95→    }
    96→    96→    
    97→    97→    // Partial match
    98→    98→    for (const [key, url] of Object.entries(SKETCHFAB_MODELS)) {
    99→    99→        if (nameLower.includes(key) || key.includes(nameLower)) {
   100→   100→            return url;
   101→   101→        }
   102→   102→    }
   103→   103→    
   104→   104→    return null;
   105→   105→}

   107→   107→// Setup 3D model viewer (or show image if no model)
   108→   108→async function setup3DModel(animal) {
   109→   109→    const modelViewer = document.getElementById('animal-3d-model');
   110→   110→    const imageElement = document.getElementById('animal-image');
   111→   111→    const toggleButton = document.getElementById('view-toggle');
   112→   112→    const toggleText = document.getElementById('view-toggle-text');
   113→   113→    
   114→   114→    if (!modelViewer || !imageElement) return;
   115→   115→    
   116→   116→    // Find matching 3D model
   117→   117→    const modelUrl = findMatchingModel(animal.name);
   118→   118→    
   119→   119→    // No 3D model found - show Wikipedia image only
   120→   120→    if (!modelUrl) {
   121→   121→        console.log(`ℹ️ No 3D model for ${animal.name} - using Wikipedia image`);
   122→   122→        modelViewer.style.display = 'none';
   123→   123→        imageElement.style.display = 'block';
   124→   124→        if (toggleButton) toggleButton.style.display = 'none';
   125→   125→        return;
   126→   126→    }
   127→   127→    
   128→   128→    // 3D model found - load it
   129→   129→    try {
   130→   130→        const oembedUrl = `https://sketchfab.com/oembed?url=${encodeURIComponent(modelUrl)}&maxwidth=800`;
   131→   131→        const response = await fetch(oembedUrl);
   132→   132→        
   133→   133→        if (response.ok) {
   134→   134→            const data = await response.json();
   135→   135→            
   136→   136→            if (data.glb_url || data.gltf_url) {
   137→   137→                // Load 3D model
   138→   138→                modelViewer.src = data.glb_url || data.gltf_url;
   139→   139→                modelViewer.alt = `3D model of ${animal.name}`;
   140→   140→                modelViewer.style.display = 'block';
   141→   141→                
   142→   142→                // Hide image initially
   143→   143→                imageElement.style.display = 'none';
   144→   144→                
   145→   145→                // Show toggle button
   146→   146→                if (toggleButton) {
   147→   147→                    toggleButton.style.display = 'flex';
   148→   148→                    toggleText.textContent = 'View Image';
   149→   149→                }
   150→   150→                
   151→   151→                // Setup toggle between 3D and image
   152→   152→                setupViewToggle(modelViewer, imageElement, toggleButton, toggleText);
   153→   153→                console.log(`✅ 3D model loaded for ${animal.name}`);
   154→   154→                return;
   155→   155→            }
   156→   156→        }
   157→   157→    } catch (error) {
   158→   158→        console.error(`❌ Error loading 3D model for ${animal.name}:`, error);
   159→   159→    }
   160→   160→    
   161→   161→    // Fallback to image if model fails to load
   162→   162→    console.log(`⚠️ 3D model failed for ${animal.name} - using image`);
   163→   163→    modelViewer.style.display = 'none';
   164→   164→    imageElement.style.display = 'block';
   165→   165→    if (toggleButton) toggleButton.style.display = 'none';

   436→   436→                renderAnimalCard(animal, type);
   437→   437→                break;
   438→   438→            }
   439→   439→        }
   440→   440→    });
   441→   441→    
   442→   442→    // Update counts
   443→   443→    const counts = { mammal: 0, bird: 0, reptile: 0, fish: 0, amphibian: 0, insect: 0 };
   444→   444→    filtered.forEach(animal => {
   445→   445→        const animalType = animal.animal_type?.toLowerCase() || '';
   446→   446→        const classType = animal.classification?.class?.toLowerCase() || '';
   447→   447→        
   448→   448→        for (const [type, validTypes] of Object.entries(typeMappings)) {
   449→   449→            if (validTypes.some(t => animalType.includes(t) || classType.includes(t))) {
   450→   450→                counts[type]++;
   451→   451→                break;
   452→   452→            }
   453→   453→        }
   454→   454→    });
   455→   455→    
   456→   456→    for (const [type, count] of Object.entries(counts)) {
   457→   457→        const countElement = document.getElementById(`count-${type}`);
   458→   458→        if (countElement) {
   459→   459→            countElement.textContent = count;
   460→   460→        }
   461→   461→    }
   462→   462→}

   775→   775→function addLocationDots(locationsString) {
   776→   776→    const dotsContainer = document.getElementById('location-dots');

   795→   795→                dot.setAttribute('data-label', location);
   796→   796→                dot.setAttribute('title', location);
   797→   797→                
   798→   798→                dotsContainer.appendChild(dot);
   799→   799→            }
   800→   800→        }
   801→   801→    });

   804→   804→function findLocationCoordinates(location) {
   805→   805→    const loc = location.toLowerCase().trim();
   806→   806→    
   807→   807→    // Direct match
   808→   808→    if (locationCoordinates[loc]) {
   809→   809→        return locationCoordinates[loc];
   810→   810→    }
   811→   811→    
   812→   812→    // Partial match
   813→   813→    for (const key in locationCoordinates) {
   814→   814→        if (loc.includes(key) || key.includes(loc)) {
   815→   815→            return locationCoordinates[key];
   816→   816→        }
   817→   817→    }
   818→   818→    
   819→   819→    // Fallback guesses
   820→   820→    if (loc.includes('asia') || loc.includes('east')) return locationCoordinates['asia'];
   821→   821→    if (loc.includes('america') || loc.includes('usa') || loc.includes('us') || loc.includes('united states')) {
   822→   822→        return locationCoordinates['north america'];
   823→   823→    }
   824→   824→    if (loc.includes('europe')) return locationCoordinates['europe'];
   825→   825→    if (loc.includes('africa')) return locationCoordinates['africa'];
   826→   826→    if (loc.includes('australia') || loc.includes('oceania')) return locationCoordinates['australia'];
   827→   827→    if (loc.includes('south') && !loc.includes('korea')) return locationCoordinates['south america'];
   828→   828→    if (loc.includes('india')) return locationCoordinates['india'];
   829→   829→    if (loc.includes('china')) return locationCoordinates['china'];
   830→   830→    if (loc.includes('russia')) return locationCoordinates['russia'];
   831→   831→    if (loc.includes('indonesia')) return locationCoordinates['indonesia'];
   832→   832→    
   833→   833→    return null;
   834→   834→}

   876→   876→        width = '90%';

   880→   880→        millionsYears = 500;

   887→   887→        width = '92%';
